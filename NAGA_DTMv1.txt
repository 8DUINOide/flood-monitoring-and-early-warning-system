// 1. Load your uploaded Naga City DTM
var nagaDTM = ee.Image('projects/floodmonitor-482303/assets/Naga_DTM');

// 2. Define Naga City Area of Interest (using a point or boundary)
var nagaCity = ee.Geometry.Point([123.19, 13.62]); 
Map.centerObject(nagaDTM, 13);

// 3. Terrain Analysis: Calculate Slope
// High slope areas are unlikely to flood; helps filter out SAR shadows.
var slope = ee.Terrain.slope(nagaDTM);
var flatAreas = slope.lt(5); // Binary mask where 1 is flat, 0 is steep

// 4. Visualization Parameters
var dtmVis = {min: 0, max: 50, palette: ['green', 'yellow', 'red']};
var slopeVis = {min: 0, max: 15, palette: ['white', 'black']};

// 5. Display Layers
Map.addLayer(nagaDTM, dtmVis, 'Naga DTM Elevation');
Map.addLayer(slope, slopeVis, 'Slope (Steepness)', false);

// 6. Practical Application: Masking a Sentinel-1 Flood Map
// This is a snippet of how you would combine it with Radar data
var s1 = ee.ImageCollection('COPERNICUS/S1_GRD')
  .filterBounds(nagaCity)
  .filter(ee.Filter.listContains('transmitterReceiverPolarisation', 'VH'))
  .first();

// Mask out steep areas from the radar image to reduce errors
var maskedS1 = s1.updateMask(flatAreas);
Map.addLayer(maskedS1, {min: -25, max: 0}, 'Cleaned Sentinel-1 SAR');
